\draw[->] (d1.west) -- ++(-1.25,0) |- (q1.west);

\begin{scope}[xshift=5.4cm]

\draw[->] (aa)--(da); \draw[->] (ar)--(dr);

\draw[->] (da.west) -- ++(-0.55,0) |- (qa.west);
\draw[->] (dr.east) -- ++(0.55,0) |- (qr.east);
\node[font=\tiny,align=center,text=black!60] at (1.05,-2.45)
